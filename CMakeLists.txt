cmake_minimum_required(VERSION 3.16)
project(celeste_native_kernel LANGUAGES CXX)
option(CELESTE_VITA "Build the bounded Vita room diagnostic executable and private VPK" OFF)

add_library(celeste_native_kernel STATIC src/native/movement.cpp src/native/grid.cpp src/native/scheduling.cpp src/native/input.cpp src/native/player.cpp src/native/player_dash.cpp src/native/player_jumps.cpp src/native/player_climb.cpp src/native/hazards.cpp)
target_sources(celeste_native_kernel PRIVATE src/native/presentation.cpp src/native/player_presentation.cpp)
target_include_directories(celeste_native_kernel PUBLIC src/native)
target_compile_features(celeste_native_kernel PUBLIC cxx_std_17)
set_target_properties(celeste_native_kernel PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(celeste_native_kernel PUBLIC -fno-fast-math -ffp-contract=off)
    target_compile_options(celeste_native_kernel PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
    message(FATAL_ERROR "The kernel requires validated GNU/Clang strict floating-point flags")
endif()

include(CTest)
add_library(celeste_room_diagnostic STATIC src/vita/diagnostic.cpp src/vita/assets.cpp src/vita/frame.cpp src/vita/draw_geometry.cpp)
target_include_directories(celeste_room_diagnostic PUBLIC src/vita)
target_link_libraries(celeste_room_diagnostic PUBLIC celeste_native_kernel)
target_compile_options(celeste_room_diagnostic PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(CELESTE_VITA)

endif()
