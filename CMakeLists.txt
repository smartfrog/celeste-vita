cmake_minimum_required(VERSION 3.16)
project(celeste_native_kernel LANGUAGES CXX)

add_library(celeste_native_kernel STATIC src/native/movement.cpp)
target_include_directories(celeste_native_kernel PUBLIC src/native)
target_compile_features(celeste_native_kernel PUBLIC cxx_std_17)
set_target_properties(celeste_native_kernel PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(celeste_native_kernel PUBLIC -fno-fast-math -ffp-contract=off)
    target_compile_options(celeste_native_kernel PRIVATE -Wall -Wextra -Wpedantic -Werror)
else()
    message(FATAL_ERROR "The kernel requires validated GNU/Clang strict floating-point flags")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(native_kernel_tests tests/native/movement_tests.cpp)
    target_link_libraries(native_kernel_tests PRIVATE celeste_native_kernel)
    add_test(NAME native_kernel_unit COMMAND native_kernel_tests)
    add_executable(native_trace_runner tests/native/trace_runner.cpp)
    target_link_libraries(native_trace_runner PRIVATE celeste_native_kernel)
    add_test(NAME native_fixture_smoke COMMAND native_trace_runner "${CMAKE_CURRENT_SOURCE_DIR}/tests/native/movement.fixtures")
endif()
