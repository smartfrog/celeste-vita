set(CELESTE_VITAGL_LIBRARY "" CACHE FILEPATH "Existing no-splash vitaGL archive")
set(CELESTE_VITAGL_INCLUDE_DIR "" CACHE PATH "Matching vitaGL headers; optional when already installed in the SDK")
set(CELESTE_ROOM_STAGING "" CACHE PATH "Verified private staging from stage_native_room.py")
set(CELESTE_EXPECTED_SOURCE_STATE "" CACHE FILEPATH "Explicit complete expected code identity; absent requires committed-clean source")
if(NOT EXISTS "${CELESTE_VITAGL_LIBRARY}" OR NOT EXISTS "${CELESTE_ROOM_STAGING}/asset_pins.hpp")
    message(FATAL_ERROR "Supply the existing no-splash archive and verified private room staging")
endif()
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
add_executable(celeste_native_room app.cpp draw.cpp trail_target.cpp)
if(CELESTE_VITAGL_INCLUDE_DIR)
    if(NOT EXISTS "${CELESTE_VITAGL_INCLUDE_DIR}/vitaGL.h")
        message(FATAL_ERROR "CELESTE_VITAGL_INCLUDE_DIR must contain the matching vitaGL.h")
    endif()
    target_include_directories(celeste_native_room PRIVATE "${CELESTE_VITAGL_INCLUDE_DIR}")
endif()
option(CELESTE_ROOM_TEXTURED "Use the bounded source-baked partial textured room instead of debug hitboxes" OFF)
if(CELESTE_ROOM_TEXTURED)
    if(NOT EXISTS "${CELESTE_ROOM_STAGING}/assets/native-room/visuals.txt")
        message(FATAL_ERROR "Textured mode needs verified private visual staging")
    endif()
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_TEXTURED)
    set(visual_files FILE "${CELESTE_ROOM_STAGING}/assets/native-room/visuals.txt" assets/native-room/visuals.txt)
    foreach(index RANGE 0 5)
        list(APPEND visual_files FILE "${CELESTE_ROOM_STAGING}/assets/native-room/texture${index}.rgba" assets/native-room/texture${index}.rgba)
    endforeach()
endif()
option(CELESTE_ROOM_SPIKE_ART "Draw nineteen static source spike images instead of red diagnostic markers" OFF)
if(CELESTE_ROOM_SPIKE_ART)
    if(NOT CELESTE_ROOM_TEXTURED)
        message(FATAL_ERROR "Source spike art requires the textured room intent")
    endif()
    file(STRINGS "${CELESTE_ROOM_STAGING}/assets/native-room/visuals.txt" spike_header LIMIT_COUNT 4)
    list(GET spike_header 3 spike_textures)
    if(NOT spike_textures EQUAL 8 OR NOT EXISTS "${CELESTE_ROOM_STAGING}/assets/native-room/texture6.rgba"
       OR NOT EXISTS "${CELESTE_ROOM_STAGING}/assets/native-room/texture7.rgba")
        message(FATAL_ERROR "Source spike art needs the verified eight-texture CSRV2 staging")
    endif()
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_SPIKE_ART)
    list(APPEND visual_files FILE "${CELESTE_ROOM_STAGING}/assets/native-room/texture6.rgba" assets/native-room/texture6.rgba)
    list(APPEND visual_files FILE "${CELESTE_ROOM_STAGING}/assets/native-room/texture7.rgba" assets/native-room/texture7.rgba)
endif()
option(CELESTE_ROOM_DECALS "Draw the bounded 22 source static decal images instead of only tiles/spikes" OFF)
if(CELESTE_ROOM_DECALS)
    if(NOT CELESTE_ROOM_TEXTURED OR NOT CELESTE_ROOM_SPIKE_ART)
        message(FATAL_ERROR "Static decal art requires the textured room and nineteen source spike intents")
    endif()
    file(STRINGS "${CELESTE_ROOM_STAGING}/assets/native-room/decals.txt" decal_header LIMIT_COUNT 1)
    if(NOT decal_header STREQUAL "CSD1" OR NOT EXISTS "${CELESTE_ROOM_STAGING}/assets/native-room/decal0.rgba")
        message(FATAL_ERROR "Static decal art needs the verified CSD1 protocol and bounded page")
    endif()
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_DECALS)
    list(APPEND visual_files FILE "${CELESTE_ROOM_STAGING}/assets/native-room/decals.txt" assets/native-room/decals.txt)
    list(APPEND visual_files FILE "${CELESTE_ROOM_STAGING}/assets/native-room/decal0.rgba" assets/native-room/decal0.rgba)
endif()
option(CELESTE_ROOM_ANIMATED_PLAYER "Add the source-backed bounded animated player and simulated hair" OFF)
if(CELESTE_ROOM_ANIMATED_PLAYER)
    if(NOT CELESTE_ROOM_TEXTURED OR NOT EXISTS "${CELESTE_ROOM_STAGING}/assets/native-room/player.txt")
        message(FATAL_ERROR "Animated player requires verified private player and textured-room staging")
    endif()
    file(STRINGS "${CELESTE_ROOM_STAGING}/assets/native-room/player.txt" player_header LIMIT_COUNT 4)
    list(GET player_header 3 player_pages)
    if(player_pages LESS 1 OR player_pages GREATER 10)
        message(FATAL_ERROR "Player page count exceeds bounded upload profile")
    endif()
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_ANIMATED_PLAYER)
    list(APPEND visual_files FILE "${CELESTE_ROOM_STAGING}/assets/native-room/player.txt" assets/native-room/player.txt)
    math(EXPR player_last_page "${player_pages}-1")
    foreach(index RANGE 0 ${player_last_page})
        list(APPEND visual_files FILE "${CELESTE_ROOM_STAGING}/assets/native-room/player${index}.rgba" assets/native-room/player${index}.rgba)
    endforeach()
endif()
option(CELESTE_ROOM_CAPTURE_AFTER_WALL_JUMP "Export once after an actual input-driven WallJump instead of spawn" OFF)
option(CELESTE_ROOM_CAPTURE_AFTER_CLIMB "Export once during actual airborne drained Climb instead of spawn" OFF)
option(CELESTE_ROOM_GUEST_REPLAY "DIAGNOSTIC ONLY: replay the bounded guest-controls fixture without host input" OFF)
option(CELESTE_ROOM_GUEST_AUTOMATIC "DIAGNOSTIC ONLY: raw input fixture on the production automatic lifecycle" OFF)
option(CELESTE_ROOM_DASH_PROBE "DIAGNOSTIC ONLY: input-driven Dash and native FBO/back-buffer alpha probe" OFF)
if(CELESTE_ROOM_DASH_PROBE)
    if(CELESTE_ROOM_GUEST_AUTOMATIC OR CELESTE_ROOM_GUEST_REPLAY OR NOT CELESTE_ROOM_ANIMATED_PLAYER)
        message(FATAL_ERROR "Dash probe requires animated automatic lifecycle and excludes existing guest fixture modes")
    endif()
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_DASH_PROBE)
endif()
if(CELESTE_ROOM_GUEST_AUTOMATIC)
    if(CELESTE_ROOM_GUEST_REPLAY OR NOT CELESTE_ROOM_ANIMATED_PLAYER)
        message(FATAL_ERROR "Guest automatic requires animated Player and is exclusive with legacy guest replay")
    endif()
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_GUEST_AUTOMATIC)
endif()
if(CELESTE_ROOM_GUEST_REPLAY)
    if(NOT CELESTE_ROOM_CAPTURE_AFTER_CLIMB)
        message(FATAL_ERROR "Guest replay requires the explicit real Climb capture policy")
    endif()
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_GUEST_REPLAY)
endif()
if(CELESTE_ROOM_CAPTURE_AFTER_WALL_JUMP AND CELESTE_ROOM_CAPTURE_AFTER_CLIMB)
    message(FATAL_ERROR "Choose only one explicit deferred capture mode")
endif()
if(CELESTE_ROOM_CAPTURE_AFTER_CLIMB)
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_CAPTURE_AFTER_CLIMB)
endif()
if(CELESTE_ROOM_CAPTURE_AFTER_WALL_JUMP)
    target_compile_definitions(celeste_native_room PRIVATE CELESTE_ROOM_CAPTURE_AFTER_WALL_JUMP)
endif()
target_include_directories(celeste_native_room PRIVATE "${CELESTE_ROOM_STAGING}")
target_compile_options(celeste_native_room PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_options(celeste_native_room PRIVATE -Wl,-q -Wl,-z,nocopyreloc "-Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/layout.ld")
set_property(TARGET celeste_native_room APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/layout.ld")
target_link_libraries(celeste_native_room PRIVATE celeste_room_diagnostic celeste_command_submission
    "${CELESTE_VITAGL_LIBRARY}" vitashark SceShaccCgExt SceShaccCg_stub_weak taihen_stub_weak
    SceCtrl_stub_weak SceDisplay_stub_weak SceGxm_stub_weak SceAppMgr_stub_weak SceSysmodule_stub_weak
    SceProcessmgr_stub_weak SceRtc_stub_weak SceKernelThreadMgr_stub_weak
    SceKernelDmacMgr_stub_weak SceCommonDialog_stub_weak mathneon m stdc++)
vita_create_self(eboot.bin celeste_native_room)
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
set(binding_expected)
if(CELESTE_EXPECTED_SOURCE_STATE)
    list(APPEND binding_expected --expected-source-state "${CELESTE_EXPECTED_SOURCE_STATE}")
endif()
set(binding_source "${CMAKE_CURRENT_BINARY_DIR}/native-build-binding.cpp")
add_custom_target(native_room_binding
    COMMAND "${Python3_EXECUTABLE}" -B "${PROJECT_SOURCE_DIR}/scripts/native_room_build.py"
        --build "${PROJECT_BINARY_DIR}" --staging "${CELESTE_ROOM_STAGING}" ${binding_expected}
    BYPRODUCTS "${binding_source}" "${CMAKE_CURRENT_BINARY_DIR}/native-build-binding.json"
    VERBATIM)
add_dependencies(celeste_native_kernel native_room_binding)
add_dependencies(celeste_native_room native_room_binding)
target_sources(celeste_native_room PRIVATE "${binding_source}")
target_link_options(celeste_native_room PRIVATE -Wl,--undefined=celeste_native_build_binding)
vita_create_vpk(celeste_native_room.vpk CELR00001 eboot.bin VERSION 00.01 NAME "Celeste Native Room Diagnostic"
    FILE "${CELESTE_ROOM_STAGING}/assets/native-room/global.mask" assets/native-room/global.mask
    FILE "${CELESTE_ROOM_STAGING}/assets/native-room/hazards.txt" assets/native-room/hazards.txt
    FILE "${CELESTE_ROOM_STAGING}/assets/native-room/provenance.json" assets/native-room/provenance.json ${visual_files})
